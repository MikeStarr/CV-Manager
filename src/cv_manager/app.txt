def get_docx_text(path: str) -> str:
    parts: list[str] = []
    for p in Document(path).paragraphs:
        run_parts: list[str] = []
        for r in p.runs:
            txt = r.text
            if r.bold:
                txt = f"*{txt}*"
            if r.italic:
                txt = f"_{txt}_"
            run_parts.append(txt)
        parts.append("".join(run_parts))
    return "\n".join(parts)


def get_cv_files():
    if not os.path.exists(CV_DIR):
        os.makedirs(CV_DIR)
    return [f for f in os.listdir(CV_DIR) if f.endswith(".docx") and not f.startswith('~')]


@st.cache_data(show_spinner="Running ATS Analysis on Job Specification...")
def get_ats_keywords(job_spec: str, base_url: str, api_key: str, model: str, timeout: float, provider: str) -> dict:
    brain = CVBrain(api_key=api_key or None, base_url=base_url, model=model, timeout=timeout, provider=provider)
    return brain.extract_ats_keywords(job_spec)


def run_llm_threaded(brain, job_spec, cv_structure, cv_content, missing_keywords):
    res_queue = queue.Queue()

    def worker():
        try:
            res = brain.generate_tailored_content(
                job_spec,
                cv_structure,
                cv_content,
                return_raw=True,
                matched_keywords=missing_keywords
            )
            res_queue.put(("SUCCESS", res))
        except Exception as e:
            res_queue.put(("ERROR", e))

    t = threading.Thread(target=worker)
    t.start()
    return t, res_queue

                            <div class="apple-card-title">CV Template Match Rates</div>
                            <div class="apple-card-caption">How well each template aligns with the extracted ATS keywords.</div>
                        </div>
                        <div class="apple-pill">Scoring</div>
                    </div>
                    """,
                    unsafe_allow_html=True,
                )

                if template_scores:
                    sorted_templates = sorted(template_scores.items(), key=lambda x: x[1], reverse=True)
                    for cv_file, score in sorted_templates:
                        is_selected = (cv_file == selected_cv)
                        label = f"**{cv_file}**" + (" *(Selected)*" if is_selected else "")
                        st.markdown(f"<div class='apple-progress-label'>{label} — <strong>{score:.1f}% Match</strong></div>", unsafe_allow_html=True)
                        st.progress(score / 100.0)
                else:
                    st.write("*No scores available*")

                st.markdown("</div>", unsafe_allow_html=True)

                # Card: Gap Analysis
                if selected_cv in template_matches:
                    st.markdown('<div class="apple-card">', unsafe_allow_html=True)
                    st.markdown(
                        f"""
                        <div class="apple-card-header">
                            <div>
                                <div class="apple-card-title">Gap Analysis</div>
                                <div class="apple-card-caption">Matched and missing keywords for `{selected_cv}`.</div>
                            </div>
                            <div class="apple-pill">Gaps</div>
                        </div>
                        """,
                        unsafe_allow_html=True,
                    )

                    matches = template_matches[selected_cv]["matched"]
                    gaps = template_matches[selected_cv]["missing"]

                    col_m, col_g = st.columns(2)
                    with col_m:
                        st.success(f"✅ Matched Keywords ({len(matches)})")
                        if matches:
                            st.markdown(
                                " ".join(f"<span class='apple-chip apple-chip-kw'>{m}</span>" for m in matches),
                                unsafe_allow_html=True,
                            )
                        else:
                            st.write("None matched.")
                    with col_g:
                        st.warning(f"❌ Missing Keywords / Gaps ({len(gaps)})")
                        if gaps:
                            st.markdown(
                                " ".join(f"<span class='apple-chip apple-chip-gap'>{g}</span>" for g in gaps),
                                unsafe_allow_html=True,
                            )
                        else:
                            st.write("No gaps detected! Excellent template choice.")

                    st.markdown("</div>", unsafe_allow_html=True)

        with tab_tailor:
            st.markdown('<div class="apple-card">', unsafe_allow_html=True)
            st.markdown(
                """
                <div class="apple-card-header">
                    <div>
                        <div class="apple-card-title">Tailored CV Output</div>
                        <div class="apple-card-caption">Review changes, download the updated CV, and inspect remaining gaps.</div>
                    </div>
                    <div class="apple-pill">Result</div>
                </div>
                """,
                unsafe_allow_html=True,
            )

            if st.session_state.get('tailored_success_msg'):
                st.success(st.session_state['tailored_success_msg'])

                try:
                    with open(st.session_state['tailored_new_cv_path'], "rb") as fp:
                        data = fp.read()
                    st.markdown('<div class="apple-download-btn">', unsafe_allow_html=True)
                    st.download_button(
                        label="⬇️ Download Updated CV",
                        data=data,
                        file_name=st.session_state['tailored_new_cv_name'],
                        mime="application/vnd.openxmlformats-officedocument.wordprocessingml.document",
                        use_container_width=True
                    )
                    st.markdown('</div>', unsafe_allow_html=True)
                except Exception as e:
                    st.error(f"Error loading tailored file for download: {e}")

                st.divider()

                col_l, col_r = st.columns(2)
                with col_l:
                    st.write("#### LLM Proposed Updates (Raw)")
                    if st.session_state.get('tailored_raw_response'):
                        st.markdown('<div class="apple-mono">', unsafe_allow_html=True)
                        st.code(st.session_state['tailored_raw_response'], language="json")
                        st.markdown('</div>', unsafe_allow_html=True)
                    else:
                        st.info("No raw response details.")
                with col_r:
                    st.write("#### Text Changes (Unified Diff)")
                    if st.session_state.get('tailored_diff'):
                        st.markdown('<div class="apple-mono">', unsafe_allow_html=True)
                        st.code(st.session_state['tailored_diff'], language="diff")
                        st.markdown('</div>', unsafe_allow_html=True)
                    else:
                        st.info("No text changes detected.")

                st.divider()
                st.write("#### ⚠️ Remaining Keyword Gaps")
                remaining_gaps = st.session_state.get('tailored_remaining_gaps', [])
                if remaining_gaps:
                    st.warning(
                        f"The tailored CV still lacks mentions of {len(remaining_gaps)} ATS keywords. "
                        "You may want to address these manually or as transferable skills."
                    )
                    st.markdown(
                        " ".join(f"<span class='apple-chip apple-chip-gap'>{g}</span>" for g in remaining_gaps),
                        unsafe_allow_html=True,
                    )
                else:
                    st.success("🎉 All ATS keywords appear to be represented in the tailored CV!")
            else:
                st.info("Click **Generate Tailored CV** in the left column to produce a tailored version of your selected template.")

            st.markdown("</div>", unsafe_allow_html=True)


if __name__ == "__main__":
    main()
